# =====================================================================
# PROLINKNET - konfiguracja serwera (nazwa.pl). Plik musi nazywać się .htaccess
# i leżeć w głównym katalogu strony (razem z index.html). NIE USUWAĆ.
# Poczty e-mail nie dotyczy (poczta działa na serwerach nazwa.pl niezależnie od strony).
# =====================================================================

Options -Indexes
DirectoryIndex index.html index.php
ErrorDocument 404 /404.html
AddDefaultCharset utf-8

<IfModule mod_rewrite.c>
RewriteEngine On

# 1. Jedna wersja strony: https://prolinknet.pl (bez www). Wyjątek: /.well-known/ (certyfikat SSL).
RewriteCond %{REQUEST_URI} !^/\.well-known/
RewriteCond %{HTTPS} !=on
RewriteCond %{HTTP:X-Forwarded-Proto} !=https
RewriteRule ^(.*)$ https://prolinknet.pl/$1 [R=301,L]
RewriteCond %{REQUEST_URI} !^/\.well-known/
RewriteCond %{HTTP_HOST} ^www\. [NC]
RewriteRule ^(.*)$ https://prolinknet.pl/$1 [R=301,L]

# 2. Katalogi techniczne formularza niedostępne z sieci
RewriteRule ^(_limity|inc)(/|$) - [F,L]

# 3. Stare adresy (strona z 2014 r.) -> nowe podstrony (301 = przeniesione na stałe)
RewriteRule ^index\.(html?|php)$ / [R=301,L,NC]
RewriteRule ^oferta_firmy_Prolinknet\.pdf$ /o-firmie/ [R=301,L,NC]
RewriteRule ^partners/ /o-firmie/#partnerzy [R=301,L,NC,NE]
RewriteRule ^(images|upload)/ / [R=301,L,NC]
RewriteRule ^(css|js)/ - [G,L,NC]
# adresy z wersji testowej / skróty
RewriteRule ^(radiolinie|swiatlowody|kontakt|o-firmie|polityka-prywatnosci)\.html$ /$1/ [R=301,L,NC]
RewriteRule ^konstrukcje(\.html)?/?$ /konstrukcje-stalowe/ [R=301,L,NC]
RewriteRule ^utrzymanie(\.html)?/?$ /utrzymanie-i-outsourcing/ [R=301,L,NC]
RewriteRule ^(oferta|uslugi)/?$ /#oferta [R=301,L,NC,NE]

# 4. Adresy bez końcowego ukośnika -> z ukośnikiem
RewriteRule ^(radiolinie|swiatlowody|konstrukcje-stalowe|utrzymanie-i-outsourcing|o-firmie|kontakt|polityka-prywatnosci)$ /$1/ [R=301,L]
</IfModule>

# 5. Kompresja tekstu
<IfModule mod_deflate.c>
AddOutputFilterByType DEFLATE text/html text/css application/javascript text/javascript application/json image/svg+xml text/plain application/xml application/manifest+json
</IfModule>

# 6. Pamięć podręczna: zdjęcia i czcionki na rok, JS na rok (wersja ?v= w HTML), strony HTML zawsze świeże.
<IfModule mod_expires.c>
ExpiresActive On
ExpiresByType image/webp "access plus 1 year"
ExpiresByType image/jpeg "access plus 1 year"
ExpiresByType image/png "access plus 1 year"
ExpiresByType image/svg+xml "access plus 1 year"
ExpiresByType image/x-icon "access plus 1 year"
ExpiresByType font/woff2 "access plus 1 year"
ExpiresByType application/javascript "access plus 1 year"
ExpiresByType text/javascript "access plus 1 year"
ExpiresByType text/html "access plus 0 seconds"
</IfModule>
<IfModule mod_headers.c>
<FilesMatch "\.(webp|jpe?g|png|svg|ico|woff2)$">
Header set Cache-Control "public, max-age=31536000, immutable"
</FilesMatch>
<FilesMatch "\.js$">
Header set Cache-Control "public, max-age=31536000"
</FilesMatch>
<FilesMatch "\.(html|php)$">
Header set Cache-Control "no-cache"
</FilesMatch>
Header always set X-Content-Type-Options "nosniff"
Header always set Referrer-Policy "strict-origin-when-cross-origin"
Header always set X-Frame-Options "SAMEORIGIN"
Header always set Permissions-Policy "camera=(), microphone=(), geolocation=()"
Header always set Strict-Transport-Security "max-age=31536000"
</IfModule>
AddType font/woff2 .woff2
AddType image/webp .webp
AddType application/manifest+json .webmanifest
